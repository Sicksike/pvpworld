/* MODE PICKER MOUSE DRAG + WHEEL SCROLL */ const modePicker = document.getElementById("mode-picker"); let pickerIsDragging = false; let pickerStartX = 0; let pickerScrollStart = 0; let pickerMoved = false; let ignorePickerClick = false; /* Mouse wheel scrolls the picker horizontally - 3x speed */ modePicker.addEventListener("wheel", event => { if (modePicker.scrollWidth <= modePicker.clientWidth) return; event.preventDefault(); modePicker.scrollLeft += (Math.abs(event.deltaX) > Math.abs(event.deltaY) ? event.deltaX : event.deltaY) * 3; }, { passive: false }); /* Click and drag with a mouse */ modePicker.addEventListener("pointerdown", event => { if (event.pointerType !== "mouse" || event.button !== 0) return; pickerIsDragging = true; pickerMoved = false; pickerStartX = event.clientX; pickerScrollStart = modePicker.scrollLeft; modePicker.classList.add("dragging"); }); /* Track movement globally without capturing the pointer */ window.addEventListener("pointermove", event => { if (!pickerIsDragging) return; const distance = event.clientX - pickerStartX; if (Math.abs(distance) > 5) { pickerMoved = true; } if (pickerMoved) { modePicker.scrollLeft = pickerScrollStart - distance; } }); function stopPickerDrag() { if (!pickerIsDragging) return; pickerIsDragging = false; modePicker.classList.remove("dragging"); if (pickerMoved) { ignorePickerClick = true; setTimeout(() => { ignorePickerClick = false; }, 150); } } window.addEventListener("pointerup", stopPickerDrag); window.addEventListener("pointercancel", stopPickerDrag); /* Prevent accidental clicks after dragging */ modePicker.addEventListener("click", event => { if (ignorePickerClick) { event.preventDefault(); event.stopPropagation(); } }, true);
